async function exportOfflineBundle({ modelId, webllmCdnUrl, onProgress }) {
  onProgress("Menyiapkan JSZip…");
  const JSZip = await loadJSZip();
  const zip = new JSZip();

  onProgress("Mengambil salinan runtime WebLLM…");
  const runtimeRes = await fetch(webllmCdnUrl);
  if (!runtimeRes.ok) throw new Error("Gagal mengunduh runtime WebLLM untuk dibundel");
  const runtimeText = await runtimeRes.text();
  zip.file("js/webllm-runtime.mjs", runtimeText);

  onProgress("Mengumpulkan file model dari cache browser…");
  const { manifest, zipEntries } = await collectCachedAssets(onProgress);
  if (zipEntries.length === 0) {
    throw new Error("Tidak ada file model di cache. Pastikan model sudah selesai dimuat dulu.");
  }
  for (const { path, blob } of zipEntries) {
    zip.file(path, blob);
  }

  onProgress("Menyusun konfigurasi offline…");
  const webllmModule = await import(/* webpackIgnore: true */ webllmCdnUrl);
  const offlineConfig = buildOfflineAppConfig(webllmModule.prebuiltAppConfig, modelId, manifest);
  zip.file("model-config.json", JSON.stringify(offlineConfig, null, 2));

  const offlineAppJs = `// Auto-generated shell offline untuk model "${modelId}".
import * as webllm from "./webllm-runtime.mjs";

const els = {
  statusDot: document.getElementById("status-dot"),
  statusText: document.getElementById("status-text"),

  chatLog: document.getElementById("chat-log"),

  sendBtn: document.getElementById("send-btn"),
};

function log(line) {
  els.log.textContent += line + "\\n";

}

  const div = document.createElement("div");
  div.className = "msg msg-" + role;
  div.textContent = text;
  els.chatLog.appendChild(div);
  els.chatLog.scrollTop = els.chatLog.scrollHeight;
  return div;

let history = [];
let engine;

  engine.setInitProgressCallback((report) => {

// CDN tempat runtime WebLLM diambil. Ganti versi di sini kalau perlu upgrade.
const WEBLLM_CDN_URL = "https://esm.run/@mlc-ai/web-llm";

// Supaya bundel yang didownload tidak raksasa buat pengguna biasa, daftar
// model dibatasi ke varian kecil (kira-kira di bawah ~2.5B parameter,
// kuantisasi 4-bit). Silakan tambah/kurangi ID di sini — ID harus persis
// sama dengan model_id di prebuiltAppConfig milik WebLLM.
const PREFERRED_MODEL_HINTS = [
  "Qwen2.5-0.5B",
  "Qwen2.5-1.5B",
  "Llama-3.2-1B",
  "Llama-3.2-3B",
  "Phi-3.5-mini",
  "gemma-2-2b",
  "SmolLM2-360M",
  "SmolLM2-1.7B",
  "TinyLlama-1.1B",
];

const els = {
  modelSelect: document.getElementById("model-select"),
  modelHint: document.getElementById("model-hint"),
  loadBtn: document.getElementById("load-btn"),
  statusDot: document.getElementById("status-dot"),
  statusText: document.getElementById("status-text"),
  progressTrack: document.getElementById("progress-track"),
  progressFill: document.getElementById("progress-fill"),
  log: document.getElementById("log"),
  exportBtn: document.getElementById("export-btn"),
  exportHint: document.getElementById("export-hint"),
  chatLog: document.getElementById("chat-log"),
  emptyState: document.getElementById("empty-state"),
  chatForm: document.getElementById("chat-form"),
  chatInput: document.getElementById("chat-input"),
  sendBtn: document.getElementById("send-btn"),
};

let webllm = null;
let engine = null;
let currentModelId = null;
let modelList = [];
let history = [];
let isGenerating = false;

function log(line) {
  const time = new Date().toLocaleTimeString("id-ID", { hour12: false });
  els.log.textContent += `[${time}] ${line}\n`;
  els.log.scrollTop = els.log.scrollHeight;
}

function setStatus(text, kind) {
  els.statusText.textContent = text;
  els.statusDot.className = "status-dot" + (kind ? ` ${kind}` : "");
}

function setProgress(fraction) {
  if (fraction == null) {
    els.progressTrack.hidden = true;
    return;
  }
  els.progressTrack.hidden = false;
  els.progressFill.style.width = `${Math.round(fraction * 100)}%`;
}

async function checkWebGPU() {
  if (!("gpu" in navigator)) {
    setStatus("WebGPU tidak tersedia di browser ini", "error");
    log("Browser ini tidak mengekspos navigator.gpu. Coba Chrome/Edge versi terbaru.");
    return false;
  }
  try {
    const adapter = await navigator.gpu.requestAdapter();
    if (!adapter) {
      setStatus("WebGPU ada, tapi tidak ada GPU adapter", "error");
      return false;
    }
    return true;
  } catch (err) {
    setStatus("Gagal inisialisasi WebGPU", "error");
    log(`Error WebGPU: ${err.message}`);
    return false;
  }
}

async function loadWebLLMRuntime() {
  log(`Mengambil runtime WebLLM dari CDN (${WEBLLM_CDN_URL})…`);
  webllm = await import(/* webpackIgnore: true */ WEBLLM_CDN_URL);
  log("Runtime WebLLM siap.");
  return webllm;
}

function populateModelSelect() {
  const all = webllm.prebuiltAppConfig.model_list;

  // Utamakan model kecil yang cocok dijalankan di laptop/HP biasa.
  const preferred = all.filter((m) =>
    PREFERRED_MODEL_HINTS.some((hint) => m.model_id.includes(hint))
  );
  modelList = preferred.length > 0 ? preferred : all;

  els.modelSelect.innerHTML = "";
  for (const m of modelList) {
    const opt = document.createElement("option");
    opt.value = m.model_id;
    const sizeMB = m.vram_required_MB ? ` (~${Math.round(m.vram_required_MB)} MB VRAM)` : "";
    opt.textContent = m.model_id + sizeMB;
    els.modelSelect.appendChild(opt);
  }
  els.modelSelect.disabled = false;
  els.loadBtn.disabled = false;
  updateModelHint();
}

function updateModelHint() {
  const m = modelList.find((x) => x.model_id === els.modelSelect.value);
  if (!m) return;
  const size = m.vram_required_MB ? `${Math.round(m.vram_required_MB)} MB` : "tidak diketahui";
  els.modelHint.textContent = `Perkiraan ukuran download & VRAM: ${size}. Didownload sekali, lalu dicache browser.`;
}

async function loadSelectedModel() {
  const modelId = els.modelSelect.value;
  if (!modelId) return;

  els.loadBtn.disabled = true;
  els.modelSelect.disabled = true;
  els.exportBtn.disabled = true;
  isGenerating = false;
  setStatus("Menyiapkan model…", "busy");
  setProgress(0);
  log(`Memuat model: ${modelId}`);

  try {
    if (!engine) {
      engine = new webllm.MLCEngine();
      engine.setInitProgressCallback((report) => {
        setProgress(report.progress);
        setStatus(report.text || "Mengunduh…", "busy");
      });
    }

    await engine.reload(modelId);
    currentModelId = modelId;
    history = [];

    setStatus(`Siap — ${modelId}`, "ready");
    setProgress(null);
    log(`Model "${modelId}" berhasil dimuat dan siap dipakai.`);

    els.chatInput.disabled = false;
    els.sendBtn.disabled = false;
    els.exportBtn.disabled = false;
    els.exportHint.textContent = "";
    els.emptyState.hidden = true;
  } catch (err) {
    setStatus("Gagal memuat model", "error");
    setProgress(null);
    log(`Error saat memuat model: ${err.message}`);
  } finally {
    els.loadBtn.disabled = false;
    els.modelSelect.disabled = false;
  }
}

function appendMessage(role, text) {
  const div = document.createElement("div");
  div.className = `msg msg-${role}`;
  div.textContent = text;
  els.chatLog.appendChild(div);
  els.chatLog.scrollTop = els.chatLog.scrollHeight;
  return div;
}

async function sendMessage(e) {
  e.preventDefault();
  const text = els.chatInput.value.trim();
  if (!text || !engine || isGenerating) return;

  els.chatInput.value = "";
  appendMessage("user", text);
  history.push({ role: "user", content: text });

  const assistantEl = appendMessage("assistant", "…");
  isGenerating = true;
  els.sendBtn.disabled = true;

  try {
    const stream = await engine.chat.completions.create({
      messages: history,
      stream: true,
      temperature: 0.7,
    });

    let full = "";
    for await (const chunk of stream) {
      const delta = chunk.choices?.[0]?.delta?.content || "";
      full += delta;
      assistantEl.textContent = full || "…";
      els.chatLog.scrollTop = els.chatLog.scrollHeight;
    }

    history.push({ role: "assistant", content: full });

    const stats = await engine.runtimeStatsText?.();
    if (stats) {
      const meta = document.createElement("div");
      meta.className = "msg-meta";
      meta.textContent = stats;
      assistantEl.appendChild(document.createElement("br"));
      assistantEl.appendChild(meta);
    }
  } catch (err) {
    assistantEl.textContent = `[error] ${err.message}`;
    log(`Error saat generate: ${err.message}`);
  } finally {
    isGenerating = false;
    els.sendBtn.disabled = false;
    els.chatInput.focus();
  }
}

async function handleExport() {
  if (!currentModelId) return;
  els.exportBtn.disabled = true;
  try {
    await exportOfflineBundle({
      modelId: currentModelId,
      webllmCdnUrl: WEBLLM_CDN_URL,
      onProgress: (msg) => {
        els.exportHint.textContent = msg;
        log(`[export] ${msg}`);
      },
    });
    els.exportHint.textContent = "Selesai — cek folder Downloads kamu.";
  } catch (err) {
    els.exportHint.textContent = `Gagal export: ${err.message}`;
    log(`[export] Error: ${err.message}`);
  } finally {
    els.exportBtn.disabled = false;
  }
}

// Auto-resize textarea
els.chatInput.addEventListener("input", () => {
  els.chatInput.style.height = "auto";
  els.chatInput.style.height = `${Math.min(els.chatInput.scrollHeight, 160)}px`;
});
els.chatInput.addEventListener("keydown", (e) => {
  if (e.key === "Enter" && !e.shiftKey) {
    e.preventDefault();
    els.chatForm.requestSubmit();
  }
});

els.modelSelect.addEventListener("change", updateModelHint);
els.loadBtn.addEventListener("click", loadSelectedModel);
els.chatForm.addEventListener("submit", sendMessage);
els.exportBtn.addEventListener("click", handleExport);

// ===== Boot =====
(async function init() {
  const ok = await checkWebGPU();
  if (!ok) return;

  setStatus("Mengambil runtime & daftar model…", "busy");
  try {
    await loadWebLLMRuntime();
    populateModelSelect();
    setStatus("Pilih model, lalu muat", null);
  } catch (err) {
    setStatus("Gagal mengambil runtime dari CDN", "error");
    log(`Error: ${err.message}. Cek koneksi internet lalu reload halaman.`);
  }
})();
